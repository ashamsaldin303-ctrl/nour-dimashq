   250→   250→```ts
   251→   251→// app/api/lead/route.ts
   252→   252→const LeadInput = z.object({
   253→   253→  source: z.enum(["pdp","card","valuation","en","direct"]),
   254→   254→  propertyRef: z.string().regex(/^[A-Z]{3}-\d{4}-\d{3}$/).optional(),
   255→   255→  district: z.string().max(40).optional(),
   256→   256→  type: z.string().max(20).optional(),
   257→   257→  phone: z.string().regex(/^\+?[0-9]{8,15}$/),
   258→   258→  honeypot: z.string().max(0).optional().default(""),   // filled = bot
   259→   259→});
   260→   260→export async function POST(req: Request) {
   261→   261→  const body = LeadInput.safeParse(await req.json());
   262→   262→  if (!body.success) return Response.json({ok:false}, {status:422});
   263→   263→  if (body.data.honeypot) return Response.json({ok:true});      // silent sink
   264→   264→  if (rateLimited(req))   return Response.json({ok:false}, {status:429}); // in-memory, 5/min/IP
   265→   265→  await db.lead.create({data:{...body.data, status:"new"}});
   266→   266→  await notifyTelegramOptional(body.data);                       // env-gated, never blocks
   267→   267→  return Response.json({ok:true});
   268→   268→}
   269→   269→```
